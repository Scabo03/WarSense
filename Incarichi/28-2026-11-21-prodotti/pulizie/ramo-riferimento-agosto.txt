# TEMPORANEO (incarico 27, prima parte): il collaudo completo del principale
# con lo strumentario di AGOSTO (Xcode 26.6, 17F113, lo stesso della build 27),
# per distinguere ciò che fallisce per lo strumentario nuovo da ciò che falliva
# già, e per avere l'elenco di riferimento delle avvertenze del compilatore.
# Vive soltanto sul ramo riferimento-agosto, che si cancella a misura presa.
name: riferimento-agosto
on:
  push:
    branches: [riferimento-agosto]
jobs:
  riferimento:
    runs-on: macos-26
    timeout-minutes: 60
    env:
      DEVELOPER_DIR: /Applications/Xcode_26.6.app/Contents/Developer
    steps:
      - uses: actions/checkout@v5
        with:
          fetch-depth: 0
      - name: Versioni
        run: sw_vers; xcodebuild -version; swift --version; xcrun simctl list runtimes; python3 --version
      - name: xcodegen
        run: brew install xcodegen && xcodegen --version
      - name: Collaudo completo con lo strumentario di agosto
        shell: bash
        run: |
          set +e
          ./scripts/collaudo-completo.sh > collaudo.log 2>&1
          echo "USCITA_COLLAUDO=$?"
          cat collaudo.log
          exit 0
      - name: Esito delle prove sul simulatore
        if: always()
        shell: bash
        run: |
          R=Applicazione/build/collaudo-risultati.xcresult
          if [ -d "$R" ]; then
            xcrun xcresulttool get test-results summary --path "$R" | python3 -c '
          import json,sys; d=json.load(sys.stdin)
          print({k:d.get(k) for k in ["result","totalTestCount","passedTests","failedTests","skippedTests"]})
          for f in d.get("testFailures",[]): print("FALLITA", f.get("targetName"), f.get("testIdentifierString") or f.get("testName"), "|", (f.get("failureText") or "")[:500].replace("\n"," "))
          for dv in d.get("devicesAndConfigurations",[]): print("apparecchio", dv.get("device",{}).get("deviceName"), dv.get("device",{}).get("osVersion"))'
          else
            echo "nessun fascio di risultati"
          fi
